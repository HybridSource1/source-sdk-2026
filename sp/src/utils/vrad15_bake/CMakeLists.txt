cmake_minimum_required(VERSION 3.16)
project(vrad15_bake CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "vrad15_bake must be built as 64-bit")
endif()
option(VB15_USE_OIDN "Enable Intel Open Image Denoise" ON)
find_package(embree 4 REQUIRED)
find_package(Threads REQUIRED)
add_executable(vrad15_bake vrad15_bake.cpp)
target_link_libraries(vrad15_bake PRIVATE embree Threads::Threads)
if(VB15_USE_OIDN)
  find_package(OpenImageDenoise 2 REQUIRED)
  target_compile_definitions(vrad15_bake PRIVATE VB15_USE_OIDN)
  target_link_libraries(vrad15_bake PRIVATE OpenImageDenoise)
endif()
